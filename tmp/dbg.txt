function zzDebug() {
  const ss = SpreadsheetApp.openById(DADOS_ID), out = {};
  ["FinLanc", "FinReceb", "FinArquivos", "FinRegras"].forEach((n) => { const sh = ss.getSheetByName(n); if (!sh) return; const lr = sh.getLastRow(); out[n] = lr - 1; if (lr < 2) return; const c = sh.copyTo(ss); c.setName("Backup " + n + " 07-11"); c.hideSheet(); sh.getRange(2, 1, lr - 1, sh.getLastColumn()).clearContent(); out[n + "_depois"] = sh.getLastRow() - 1; });
  PropertiesService.getScriptProperties().deleteProperty("FIN_DRV_DIA");
  Logger.log(JSON.stringify(out));
}
